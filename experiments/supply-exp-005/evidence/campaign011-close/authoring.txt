class Run:
 def __init__(self,name,n):
  self.name=name;self.source=f'checkpoint-{n}.json.gz';d=unpack(OUT/self.source);self.b=d['state'];assert hash_bundle(self.b)==d['hash'];self.initial=d['hash'];self.records=[];self.failures=[]
 @classmethod
 def resume(cls,name):
  d=json.loads((OUT/(name+'.json')).read_text());self=cls.__new__(cls);self.name=name;self.source=d['checkpoint'];self.initial=d['initial_hash'];self.records=d['records'];self.failures=d['failures'];self.b=unpack(OUT/(name+'-state.json.gz'));assert hash_bundle(self.b)==d['final_hash'];return self
 def act(self,a):
  before=self.b;cmd=auto_command(before);cmd['id']='close-'+self.name+'-'+str(before['revision']);cmd['action'].update(a)
  r=execute(before,cmd,seconds=3)
  if not r['ok']:
   assert r['state']==before;self.failures.append(dict(command=cmd,error=r.get('error'),unchanged=True));self.save();raise RuntimeError(self.failures[-1])
  self.b=r['state'];j=self.b['journal'][-1];check_losses(before,self.b,j)
  old={u['id']:u for u in before['logistics']['units']};new={u['id']:u for u in self.b['logistics']['units']}
  changes=[dict(unit=u,before=old.get(u),after=new.get(u)) for u in sorted(set(old)|set(new)) if old.get(u)!=new.get(u)]
  rec=dict(command=cmd,hash=hash_bundle(self.b),turn=before['core']['turn'],phase=before['core']['phase'],seconds=r['seconds'],events=j['events'],supply_events=j['supply_events'],charges=j['charges'],settled=j['settled'],changes=changes,audit=audit(self.b['logistics']))
  if j['settled']:rec['ledger']=deepcopy(self.b['logistics']['ledger'])
  if a['type'] in ['ATTACK','SCHWERPUNKT_ATTACK']:rec['pre_action_effects']=effects(before['logistics'],cmd['action'])
  self.records.append(rec);self.save();print(self.name,self.b['revision'],a['type'],self.b['core']['phase'],self.b['core']['pendingDecision'] and self.b['core']['pendingDecision']['kind'],flush=True);return rec
 def save(self):
  packed(OUT/(self.name+'-state.json.gz'),self.b)
  (OUT/(self.name+'.json')).write_text(json.dumps(dict(base=BASE,checkpoint=self.source,initial_hash=self.initial,records=self.records,failures=self.failures,final_hash=hash_bundle(self.b)),ensure_ascii=False,indent=2))
 def choices(self,advance=False):
  while self.b['core']['pendingDecision']:
   p=self.b['core']['pendingDecision'];kind=p['kind'];base=dict(battleId=p['battleId'])
   if kind=='ADVANCE_AFTER_COMBAT':
    a=dict(type='ADVANCE_AFTER_COMBAT',unitId='G-PZ-01' if 'G-PZ-01' in p['eligibleUnitIds'] else p['eligibleUnitIds'][0],**base) if advance else dict(type='PASS_ADVANCE',**base)
   elif kind in ['BREAKTHROUGH_OPTION','SCHWERPUNKT_OPTION','DEFENDER_REACTION']:
    a=dict(type={'BREAKTHROUGH_OPTION':'PASS_BREAKTHROUGH','SCHWERPUNKT_OPTION':'PASS_SCHWERPUNKT','DEFENDER_REACTION':'PASS_REACTION'}[kind],**base)
   else:
    assert kind in ['LOSS_ALLOCATION','RETREAT']
    # Existing forced-choice helper never probes ATTACK in these two pending kinds.
    options=json.loads(subprocess.check_output(['node',str(ROOT/'choice-probe.mjs')],input=json.dumps(self.b['core']).encode()));assert options;a=options[0]
   self.act(a)
 def end(self):
  assert not self.b['core']['pendingDecision']
  if self.b['core']['phase']=='SOVIET_REINFORCEMENT_SUPPLY':
   md=metadata(self.b['core'],'S')
   for slot in md['reinforcements']:
    # Own publicly known occupied entries: skip obvious full stacks in test input.
    h=next((h for h in md['entry_hexes'] if sum(u['alive'] and u['hex']==h for u in self.b['core']['units'].values())<2),None)
    if h is not None:self.act(dict(type='DEPLOY_REINFORCEMENT',reinforcementId=slot['id'],entryHex=h))
  return self.act(dict(type='END_PHASE'))
 def until(self,turn,phase):
  for _ in range(40):
   if self.b['core']['turn']==turn and self.b['core']['phase']==phase:return
   self.end()
  raise RuntimeError('phase target not reached')
